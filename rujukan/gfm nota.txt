     <div class="note-box"><strong>A. Rumus Skala</strong>
      <div class="formula">
       Skala = Ukuran Lukisan : Ukuran Sebenar (1:n)
      </div>
      <ul class="bullet-list">
       <li><strong>1:n</strong> bermaksud 1 unit pada lukisan mewakili n unit sebenar</li>
       <li>Contoh: Skala 1:100 → 1 cm pada lukisan = 100 cm sebenar</li>
      </ul>
     </div>
     <div class="note-box"><strong>B. Jenis Skala</strong>
      <ul class="bullet-list">
       <li><strong>n &gt; 1 (Pengecilan):</strong> Objek sebenar lebih besar daripada lukisan</li>
       <li>Contoh: 1:50, 1:100, 1:1000 → Lukisan lebih kecil</li>
       <li><strong>n &lt; 1 (Pembesaran):</strong> Objek sebenar lebih kecil daripada lukisan</li>
       <li>Contoh: 1:1/2, 2:1, 10:1 → Lukisan lebih besar</li>
      </ul>
     </div>
     <div class="note-box warning"><strong>⚠️ AMARAN PENTING - PERANGKAP LUAS</strong> <span class="warning-badge">SERING KELUAR!</span>
      <ul class="bullet-list">
       <li><strong>Jika skala panjang ialah 1:n</strong></li>
       <li><strong>Maka nisbah LUAS ialah 1:n²</strong></li>
       <li><strong>JANGAN sekadar darab n untuk luas!</strong></li>
       <li>Contoh: Skala 1:10 untuk panjang → Skala luas adalah 1:100</li>
       <li>Formula: Luas Sebenar = Luas Lukisan × n²</li>
      </ul>

     <div class="section-title"><span class="section-icon">📝</span> <span>1. NOTA PENTING (HIGH YIELD)</span>
     </div>
     <div class="note-box"><strong>A. Teorem Pythagoras</strong>
      <div class="formula">
       c² = a² + b²
      </div>
      <ul class="bullet-list">
       <li>c = Hipotenus (sisi terpanjang, bertentangan dengan sudut tegak)</li>
       <li>a, b = Sisi-sisi lain (kaki segi tiga)</li>
       <li><strong>Ingat:</strong> Formula ini hanya untuk segi tiga bersudut tegak (90°)</li>
      </ul>
     </div>
     <div class="note-box"><strong>B. Nisbah Trigonometri - SOHCAHTOA</strong>
      <div class="formula">
       SOH: Sin θ = Tentangan / Hipotenus
      </div>
      <div class="formula">
       CAH: Kos θ = Sebelahan / Hipotenus
      </div>
      <div class="formula">
       TOA: Tan θ = Tentangan / Sebelahan
      </div>
      <ul class="bullet-list">
       <li><strong>Tentangan (Opposite):</strong> Sisi bertentangan dengan sudut θ</li>
       <li><strong>Sebelahan (Adjacent):</strong> Sisi bersebelahan dengan sudut θ</li>
       <li><strong>Hipotenus (Hypotenuse):</strong> Sisi terpanjang (bertentangan sudut 90°)</li>
       <li><strong>PENTING:</strong> Pastikan kalkulator dalam mod "DEGREE" (bukan radian)</li>
      </ul>
     </div>
     <div class="note-box"><strong>C. Nilai Trigonometri Istimewa</strong>
      <ul class="bullet-list">
       <li>sin 0° = 0, sin 30° = 0.5, sin 45° = 0.707, sin 90° = 1</li>
       <li>cos 0° = 1, cos 30° = 0.866, cos 60° = 0.5, cos 90° = 0</li>
       <li>tan 0° = 0, tan 45° = 1, tan 90° = tidak terdefinisi (∞)</li>
      </ul>
     </div>
     <div class="note-box warning"><strong>D. Graf Trigonometri</strong>
      <ul class="bullet-list">
       <li><strong>Graf Sin:</strong> Bermula dari 0 (titik 0°), bentuk gelombang naik</li>
       <li><strong>Graf Kos:</strong> Bermula dari 1 (titik tertinggi pada 0°), bentuk turun</li>
       <li><strong>Ingat:</strong> Sin bermula dari tengah (0), Kos bermula dari puncak (1)</li>

     <h2 class="module-header" id="module4Title">📊 MODUL 4: STATISTIK</h2>
     <div class="section-title"><span class="section-icon">📝</span> <span>1. NOTA PENTING (HIGH YIELD)</span>
     </div>
     <div class="note-box"><strong>A. Kecenderungan Memusat (Measures of Central Tendency)</strong>
      <ul class="bullet-list">
       <li><strong>Min (Purata):</strong> Jumlah semua nilai ÷ Bilangan data</li>
       <li>Formula: Min = Σx / n</li>
       <li><strong>Median (Nilai Tengah):</strong> Nilai di tengah-tengah apabila data disusun</li>
       <li>Jika n ganjil: Median = nilai tengah</li>
       <li>Jika n genap: Median = (nilai ke-n/2 + nilai ke-(n/2+1)) / 2</li>
       <li><strong>Mod (Nilai Paling Kerap):</strong> Nilai dengan kekerapan tertinggi</li>
      </ul>
     </div>
     <div class="note-box"><strong>B. Julat dan Serakan</strong>
      <ul class="bullet-list">
       <li><strong>Julat (Range):</strong> Nilai Terbesar - Nilai Terkecil</li>
       <li><strong>Varians:</strong> Ukuran penyebaran data dari min</li>
       <li><strong>Sisihan Piawai:</strong> √Varians (ukuran serakan yang lebih mudah)</li>
      </ul>
     </div>
     <div class="note-box"><strong>C. Graf Statistik</strong>
      <ul class="bullet-list">
       <li><strong>Histogram:</strong> Bar chart untuk data berkumpulan</li>
       <li>Kelas Mod = Bar tertinggi (kekerapan paling banyak)</li>
       <li>Paksi-X: Kelas/Selang, Paksi-Y: Kekerapan</li>
       <li><strong>Ogif (Graf Kekerapan Longgokan):</strong> Graf berbentuk S</li>
       <li>Digunakan untuk cari Median dan Kuartil</li>
       <li>Paksi-Y: Kekerapan Longgokan (cumulative frequency)</li>
      </ul>
     </div>
     <div class="note-box warning"><strong>D. Cari Median dari Ogif</strong> <span class="warning-badge">TEKNIK PENTING!</span>
      <ul class="bullet-list">
       <li>Kedudukan Median = n/2 (n = jumlah data)</li>
       <li>Contoh: Jika n = 40, maka Median pada kedudukan ke-20</li>
       <li>Baca nilai dari Ogif pada kekerapan longgokan = 20</li>
      </ul>

        function renderQuestion() {
            const questions = quizData[currentQuizModule];
            const currentQuestion = questions[currentQuestionIndex];
            const totalQuestions = questions.length;
            const progress = ((currentQuestionIndex + 1) / totalQuestions) * 100;

            const moduleNames = {
                1: "Modul 1: Pengukuran Asas",
                2: "Modul 2: Lukisan Berskala",
                3: "Modul 3: Trigonometri",
                4: "Modul 4: Statistik"
            };

            let html = `
                <div class="quiz-container">
                    <div class="quiz-header">
                        <h2 style="color: #667eea; margin: 0 0 10px 0;">🎮 ${moduleNames[currentQuizModule]}</h2>
                        <p style="color: #666; margin: 0;">Soalan ${currentQuestionIndex + 1} daripada ${totalQuestions}</p>
                        <div class="quiz-progress">
                            <div class="quiz-progress-bar" style="width: ${progress}%"></div>
                        </div>
                    </div>

                    <div class="question-card">
                        <div class="question-number">Soalan ${currentQuestionIndex + 1}</div>
                        <div class="question-text">${currentQuestion.question}</div>
                        <div class="options-container" id="optionsContainer">
                            ${currentQuestion.options.map((option, index) => `
                                <button class="option-button" onclick="selectAnswer(${index})" id="option${index}">
                                    <span class="option-label">${String.fromCharCode(65 + index)}</span>
                                    <span>${option}</span>
                                </button>
                            `).join('')}
                        </div>
                        <div class="explanation-box" id="explanationBox">
                            <div class="explanation-title" id="explanationTitle"></div>
                            <div class="explanation-text" id="explanationText"></div>
                        </div>
                    </div>

                    <div class="quiz-actions">
                        <button class="action-button" id="nextButton" onclick="nextQuestion()" style="display: none;">
                            Soalan Seterusnya →
                        </button>
                    </div>
                </div>
            `;

            document.getElementById('quizContent').innerHTML = html;
            selectedAnswer = null;
        }

        function selectAnswer(answerIndex) {
            if (selectedAnswer !== null) return;

            selectedAnswer = answerIndex;
            const questions = quizData[currentQuizModule];
            const currentQuestion = questions[currentQuestionIndex];
            const isCorrect = answerIndex === currentQuestion.correct;

            if (isCorrect) {
                quizScore++;
            }

            // Show result on selected option
            const selectedOption = document.getElementById('option' + answerIndex);
            selectedOption.classList.add(isCorrect ? 'correct' : 'wrong');

            // Show correct answer if wrong
            if (!isCorrect) {
                document.getElementById('option' + currentQuestion.correct).classList.add('correct');
            }

            // Disable all options
            for (let i = 0; i < currentQuestion.options.length; i++) {
                document.getElementById('option' + i).classList.add('disabled');
            }

            // Show explanation
            const explanationBox = document.getElementById('explanationBox');
            const explanationTitle = document.getElementById('explanationTitle');
            const explanationText = document.getElementById('explanationText');

            explanationTitle.innerHTML = isCorrect 
                ? '✅ Betul! Tahniah!' 
                : '❌ Salah. Jawapan yang betul ialah ' + String.fromCharCode(65 + currentQuestion.correct);
            explanationText.textContent = currentQuestion.explanation;
            explanationBox.classList.add('show');

            // Show next button
            document.getElementById('nextButton').style.display = 'block';
        }

        function nextQuestion() {
            const questions = quizData[currentQuizModule];
            currentQuestionIndex++;

            if (currentQuestionIndex < questions.length) {
                renderQuestion();
            } else {
                showQuizResults();
            }
        }

        function showQuizResults() {
            const questions = quizData[currentQuizModule];
            const totalQuestions = questions.length;
            const percentage = Math.round((quizScore / totalQuestions) * 100);

            let message = '';
            let emoji = '';
            if (percentage >= 80) {
                message = 'Cemerlang! Teruskan usaha! 🌟';
                emoji = '🏆';
            } else if (percentage >= 60) {
                message = 'Bagus! Terus berusaha! 💪';
                emoji = '👍';
            } else {
                message = 'Cuba lagi! Anda boleh buat lebih baik! 📚';
                emoji = '💡';
            }

            const moduleNames = {
                1: "Modul 1: Pengukuran Asas",
                2: "Modul 2: Lukisan Berskala",
                3: "Modul 3: Trigonometri",
                4: "Modul 4: Statistik"
            };

            let html = `
                <div class="quiz-container">
                    <div class="quiz-score">
                        <div style="font-size: 48px; margin-bottom: 10px;">${emoji}</div>
                        <h2 style="margin: 10px 0;">Tamat!</h2>
                        <h3 style="margin: 5px 0; font-weight: 400;">${moduleNames[currentQuizModule]}</h3>
                        <div class="score-number">${quizScore}/${totalQuestions}</div>
                        <div style="font-size: 32px; margin: 10px 0;">${percentage}%</div>
                        <div class="score-message">${message}</div>
                    </div>

                    <div class="quiz-actions">
                        <button class="action-button primary" onclick="startQuiz(${currentQuizModule})">
                            🔄 Cuba Lagi
                        </button>
                        <button class="action-button" onclick="exitQuiz()">
                            ← Kembali ke Modul
                        </button>
                    </div>
                </div>
            `;

            document.getElementById('quizContent').innerHTML = html;
        }

        function exitQuiz() {
            showAllContent();
        }
