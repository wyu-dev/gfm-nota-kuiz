            const questions = questionsData[currentTopic];
            const exerciseContent = document.getElementById('exerciseContent');
            exerciseContent.innerHTML = '';

                const questionDiv = document.createElement('div');
                questionDiv.className = 'question-item bg-white rounded-xl shadow-md p-6 mb-6';
                questionDiv.innerHTML = `
                    <h4 class="text-lg font-bold text-gray-800 mb-3">Soalan ${index + 1}</h4>
                    <p class="text-gray-700 mb-4">${q.question}</p>
                    <div class="space-y-2" id="options-${index}">

                            <div class="answer-option border-2 border-gray-300 rounded-lg p-3 cursor-pointer"

                                <span class="font-semibold">${String.fromCharCode(65 + optIndex)}.</span> ${opt}

        function selectAnswer(questionIndex, selectedIndex, correctIndex) {
            if (answeredQuestions.includes(questionIndex)) return;
            
            answeredQuestions.push(questionIndex);
            const optionsDiv = document.getElementById(`options-${questionIndex}`);
            const options = optionsDiv.querySelectorAll('.answer-option');
            
            options.forEach((opt, idx) => {
                opt.style.cursor = 'not-allowed';
                opt.onclick = null;
                
                if (idx === correctIndex) {
                    opt.classList.add('correct');
                }
                if (idx === selectedIndex && selectedIndex !== correctIndex) {
                    opt.classList.add('wrong');
                }
            });
            
            if (selectedIndex === correctIndex) {
                currentScore++;
            }
            
            const explanationDiv = document.getElementById(`explanation-${questionIndex}`);
            explanationDiv.classList.remove('hidden');
            explanationDiv.className = selectedIndex === correctIndex 
                ? 'mt-4 p-4 rounded-lg bg-green-50 border-l-4 border-green-500'
                : 'mt-4 p-4 rounded-lg bg-red-50 border-l-4 border-red-500';
            explanationDiv.innerHTML = `
                <p class="font-semibold ${selectedIndex === correctIndex ? 'text-green-800' : 'text-red-800'} mb-2">
                    ${selectedIndex === correctIndex ? '✓ Betul!' : '✗ Salah'}
                </p>
                <p class="text-gray-700">${questionsData[currentTopic][questionIndex].explanation}</p>
            `;
            
            document.getElementById('scoreDisplay').textContent = `${currentScore}/10`;
            document.getElementById('questionProgress').textContent = `${answeredQuestions.length}/10`;
            
            if (answeredQuestions.length === 10) {
                showResult();
            }
        }

            }, 500);
        }
