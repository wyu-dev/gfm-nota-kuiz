        // Toggle module visibility
        function toggleModule(moduleId) {
            const content = document.getElementById(moduleId);
            const icon = document.getElementById(moduleId + '-icon');
            
            if (content.classList.contains('hidden')) {
                content.classList.remove('hidden');
                icon.classList.add('rotate-180');
            } else {
                content.classList.add('hidden');
                icon.classList.remove('rotate-180');
            }
        }
        
        // Check answer and show immediate feedback
        async function checkAnswer(questionId, selectedAnswer, correctAnswer, moduleId) {
            if (!isQuizStarted) {
                showToast('❌ Sila klik "Mula Jawab" terlebih dahulu!', 'error');
                return;
            }

            const question = questionsDB.find(q => q.id === questionId);
            const feedbackDiv = document.getElementById(`feedback-${questionId}`);
            const feedbackContent = document.getElementById(`feedback-content-${questionId}`);
            const isCorrect = selectedAnswer === correctAnswer;
            
            // Disable all options for this question after first selection
            const radios = document.querySelectorAll(`input[name="q${questionId}"]`);
            radios.forEach(radio => {
                radio.disabled = true;
            });
            
            // Update option labels styling
            const labels = document.querySelectorAll(`label[data-question="${questionId}"]`);
            labels.forEach(label => {
                const option = label.getAttribute('data-option');
                
                // Reset styles
                label.classList.remove('border-green-500', 'bg-green-50', 'border-red-500', 'bg-red-50');
                
                if (option === correctAnswer) {
                    // Highlight correct answer in green
                    label.classList.add('border-green-500', 'bg-green-50', 'border-2');
                } else if (option === selectedAnswer && !isCorrect) {
                    // Highlight wrong answer in red
                    label.classList.add('border-red-500', 'bg-red-50', 'border-2');
                }
            });
            
            // Show feedback
            feedbackDiv.classList.remove('hidden');
            
            if (isCorrect) {
                feedbackDiv.className = 'mt-4 p-5 rounded-lg bg-green-50 border-2 border-green-500 animate-slide-in';
                feedbackContent.innerHTML = `

                            </svg>
                        </div>
                        <div class="flex-1">
                            <h4 class="text-lg font-bold text-green-800 mb-2">✅ Tahniah! Jawapan Betul!</h4>

                feedbackDiv.className = 'mt-4 p-5 rounded-lg bg-red-50 border-2 border-red-500 animate-slide-in';

                    <div class="flex items-start space-x-3">

                            <svg class="w-8 h-8 text-red-600" fill="currentColor" viewBox="0 0 20 20">

            if (allRecords.length >= 999) {
                showToast('⚠️ Had maksimum 999 rekod telah dicapai.', 'error');
            } else {
                showLoading(true);
                const record = {
                    student_name: studentName,
                    question_number: questionId,
                    selected_answer: selectedAnswer,
                    correct_answer: correctAnswer,
                    is_correct: isCorrect,
                    module: question.module,
                    timestamp: new Date().toISOString(),
                    attempt_id: currentAttemptId
                };

                const result = await window.dataSdk.create(record);
                showLoading(false);
                
                if (!result.isOk) {
                    showToast('❌ Gagal simpan jawapan. Sila cuba lagi.', 'error');
                }
            }
            
            // Update module progress
            updateModuleProgress(moduleId);
        }
        
        // Update module progress counter
        function updateModuleProgress(moduleId) {
            const moduleContent = document.getElementById(moduleId);
            const allQuestions = moduleContent.querySelectorAll('.question-item');
            let answeredCount = 0;
            
            allQuestions.forEach(questionDiv => {
                const radios = questionDiv.querySelectorAll('input[type="radio"]');
                const isAnswered = Array.from(radios).some(radio => radio.checked);
                if (isAnswered) answeredCount++;
            });
            
            const progressEl = document.getElementById(moduleId + '-progress');
            progressEl.textContent = `${answeredCount}/${allQuestions.length}`;
            
            // Change color when complete
            if (answeredCount === allQuestions.length) {
                progressEl.classList.add('bg-green-500');
                progressEl.classList.remove('bg-white', 'bg-opacity-20');
            } else {
                progressEl.classList.remove('bg-green-500');
                progressEl.classList.add('bg-white', 'bg-opacity-20');
            }
        }

        // Start quiz
        function startQuiz() {
            const nameInput = document.getElementById('student-name');
            studentName = nameInput.value.trim();
            
            if (!studentName) {
                showToast('❌ Sila masukkan nama anda terlebih dahulu!', 'error');
                return;
            }

            isQuizStarted = true;
            currentAttemptId = 'attempt_' + Date.now();
            
            // Enable all radio buttons
            document.querySelectorAll('input[type="radio"]').forEach(radio => {
                radio.disabled = false;
            });
            
            showToast('✅ Selamat menjawab, ' + studentName + '!', 'success');
            
            // Scroll to first question
            const firstQuestion = document.querySelector('.question-item');
            if (firstQuestion) {
                firstQuestion.scrollIntoView({ behavior: 'smooth', block: 'center' });
            }
        }

        // Reset quiz
        function resetQuiz() {
            document.querySelectorAll('input[type="radio"]').forEach(radio => {
                radio.checked = false;
                radio.disabled = true;
            });
            
            // Hide all feedback
            document.querySelectorAll('[id^="feedback-"]').forEach(feedback => {
                feedback.classList.add('hidden');
            });
            
            // Reset option styling
            document.querySelectorAll('.option-label').forEach(label => {
                label.classList.remove('border-green-500', 'bg-green-50', 'border-red-500', 'bg-red-50');
            });
            
            document.getElementById('student-name').value = '';
            isQuizStarted = false;
            
            // Reset all module progress
            for (let i = 1; i <= 4; i++) {
                const modId = `module-${i}`;
                updateModuleProgress(modId);
            }
        }

        // Update statistics
        function updateStatistics() {
            const totalAttempts = new Set(allRecords.map(r => r.attempt_id)).size;
            const correctAnswers = allRecords.filter(r => r.is_correct).length;
            const wrongAnswers = allRecords.filter(r => !r.is_correct).length;
            const percentage = allRecords.length > 0 ? ((correctAnswers / allRecords.length) * 100).toFixed(1) : 0;

            document.getElementById('stat-attempts').textContent = totalAttempts;
            document.getElementById('stat-correct').textContent = correctAnswers;
            document.getElementById('stat-wrong').textContent = wrongAnswers;
            document.getElementById('stat-percentage').textContent = percentage + '%';

            // Module breakdown
            const modules = {
                "Modul 1: Pengukuran Asas": { correct: 0, total: 0 },
                "Modul 2: Lukisan Berskala": { correct: 0, total: 0 },
                "Modul 3: Trigonometri": { correct: 0, total: 0 },
                "Modul 4: Statistik": { correct: 0, total: 0 }
            };

            allRecords.forEach(r => {
                if (modules[r.module]) {
                    modules[r.module].total++;
                    if (r.is_correct) modules[r.module].correct++;
                }
            });

            const breakdownHTML = Object.keys(modules).map(modName => {
                const mod = modules[modName];
                const perc = mod.total > 0 ? ((mod.correct / mod.total) * 100).toFixed(1) : 0;
                const color = perc >= 70 ? 'green' : perc >= 50 ? 'yellow' : 'red';
                
                return `
                    <div class="border-l-4 border-${color}-500 pl-4 py-3">
                        <div class="flex justify-between items-center mb-2">
                            <h4 class="font-bold text-gray-800">${modName}</h4>
                            <span class="text-sm font-semibold text-${color}-600">${perc}%</span>
                        </div>
                        <div class="w-full bg-gray-200 rounded-full h-2">
                            <div class="bg-${color}-500 h-2 rounded-full" style="width: ${perc}%"></div>
                        </div>
                        <p class="text-xs text-gray-600 mt-1">${mod.correct}/${mod.total} betul</p>
                    </div>
                `;
            }).join('');

            document.getElementById('module-breakdown').innerHTML = breakdownHTML;
        }

        // Render results table
        function renderResultsTable() {
            const tbody = document.getElementById('results-table');
            const sortedRecords = [...allRecords].sort((a, b) => 
                new Date(b.timestamp) - new Date(a.timestamp)
            ).slice(0, 50); // Show latest 50

            if (sortedRecords.length === 0) {
                tbody.innerHTML = `
                    <tr>
                        <td colspan="6" class="px-4 py-8 text-center text-gray-500">
                            Tiada rekod lagi. Mula jawab soalan untuk melihat keputusan.
                        </td>
                    </tr>
                `;
                return;
            }

            tbody.innerHTML = sortedRecords.map(r => {
                const date = new Date(r.timestamp);
                const timeStr = date.toLocaleString('ms-MY', { 
                    day: '2-digit', 
                    month: 'short', 
                    hour: '2-digit', 
                    minute: '2-digit' 
                });

                return `
                    <tr class="hover:bg-gray-50">
                        <td class="px-4 py-3 text-sm font-medium text-gray-800">${r.student_name}</td>
                        <td class="px-4 py-3 text-sm text-gray-600">Soalan ${r.question_number}</td>
                        <td class="px-4 py-3 text-xs text-gray-500">${r.module.split(':')[0]}</td>
                        <td class="px-4 py-3 text-sm">
                            <span class="font-semibold">${r.selected_answer}</span>
                            ${r.selected_answer !== r.correct_answer ? `<span class="text-gray-400"> (✓${r.correct_answer})</span>` : ''}
                        </td>
                        <td class="px-4 py-3">
                            ${r.is_correct ? 
                                '<span class="bg-green-100 text-green-800 text-xs font-semibold px-2 py-1 rounded">✓ Betul</span>' : 
                                '<span class="bg-red-100 text-red-800 text-xs font-semibold px-2 py-1 rounded">✗ Salah</span>'
                            }
                        </td>
                        <td class="px-4 py-3 text-xs text-gray-500">${timeStr}</td>
                    </tr>
                `;
            }).join('');
        }

        // Switch tabs
        function switchTab(tab) {
            // Hide all tabs
            document.getElementById('quiz-tab').classList.add('hidden');
            document.getElementById('results-tab').classList.add('hidden');

            // Remove active class from all tab buttons
            document.getElementById('tab-quiz').classList.remove('tab-active');
            document.getElementById('tab-results').classList.remove('tab-active');

            // Show selected tab
            if (tab === 'quiz') {
                document.getElementById('quiz-tab').classList.remove('hidden');
                document.getElementById('tab-quiz').classList.add('tab-active');
            } else if (tab === 'results') {
                document.getElementById('results-tab').classList.remove('hidden');
                document.getElementById('tab-results').classList.add('tab-active');
                updateStatistics();
            }

            // Scroll to top
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Toast notification
        function showToast(message, type = 'success') {
            const toast = document.getElementById('toast');
            const icon = document.getElementById('toast-icon');
            const messageEl = document.getElementById('toast-message');
            const toastDiv = toast.querySelector('div');

            messageEl.textContent = message;

            if (type === 'success') {
                icon.textContent = '✅';
                toastDiv.style.borderLeftColor = '#10b981';
            } else if (type === 'error') {
                icon.textContent = '❌';
                toastDiv.style.borderLeftColor = '#ef4444';
            }

            toast.classList.remove('hidden');

            setTimeout(() => {
                toast.classList.add('hidden');
            }, 4000);
        }

                spinner.classList.add('flex');
            } else {
                spinner.classList.add('hidden');
                spinner.classList.remove('flex');
            }
        }
